cmake_minimum_required(VERSION 3.16)
project(independent_dd LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
add_executable(dd src/main.cpp)
target_compile_options(dd PRIVATE -O3 -Wall -Wextra -Wpedantic)
target_link_libraries(dd PRIVATE OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(dd PRIVATE stdc++fs)
endif()
