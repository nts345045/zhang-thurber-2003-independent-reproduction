double segment(const Grid &g, P a, P b, P *ga = nullptr, P *gb = nullptr,
               Vec *dv = nullptr, Vec *length = nullptr) {
  P d = b - a;
  double L = norm(d);
  if (L < 1e-12)
    return 0;
  Vec cuts{0, 1};
  const Vec *axes[] = {&g.x, &g.y, &g.z};
  for (int k = 0; k < 3; k++)
    if (std::abs(d[k]) > 1e-12)
      for (double q : *axes[k]) {
        double t = (q - a[k]) / d[k];
        if (t > 1e-10 && t < 1 - 1e-10)
          cuts.push_back(t);
      }
  std::sort(cuts.begin(), cuts.end());
